"""A tiny tool for weaving two words into something new."""

import argparse

def weave(word1: str, word2: str) -> str:
    """Weave two words together character by character."""
    # Find the shorter word length
    min_len = min(len(word1), len(word2))

if __name__ == "__main__":

#!/usr/bin/env python3
"""
Word Weaver

A small script to weave two words together into a new seed.
Each character is taken from one word alternately.

Usage:
    python word_weaver.py quiet now
    Output: qnuoiwet
"""

import sys

def weave(word1, word2):
    result = []
    # Alternate characters: first from word1, then word2, and so on
    min_len = min(len(word1), len(word2))
    for i in range(min_len):
        result.append(word1[i])
        result.append(word2[i])
    # Add remaining characters from the longer word
    result.extend(word1[min_len:])
    result.extend(word2[min_len:])
    return "".join(result)

def main():
    if len(sys.argv) < 3:
        print("Usage: python word_weaver.py <word1> <word2>")
        print("Example: python word_weaver.py quiet now")
        sys.exit(1)

    word1 = sys.argv[1].lower()
    word2 = sys.argv[2].lower()
    seed = weave(word1, word2)
    print(seed)

if __name__ == "__main__":
    main()
